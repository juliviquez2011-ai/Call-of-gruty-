this.physics.add.overlap(player, enemies, onPlayerHit, null, this); // ----- HUD ----- scoreText = this.add.text(10, 10, 'Puntos: 0', { font: '20px monospace', fill: '#fff' }); // ----- SONIDO ----- laserSound = this.sound.add('laser'); } // ------------------------------------------------- function update(time, delta) { // ---- MOVIMIENTO PLAYER ---- const speed = 200; player.setVelocity(0); if (cursors.left.isDown) player.setVelocityX(-speed); if (bullet) { bullet.enableBody(true, player.x, player.y - 20, true, true) .setVelocityY(-400) .setTexture('bullet') .setScale(0.5); laserSound.play({ volume: 0.2 }); } } // ---- SPAWN ENEMIGOS ---- if (time > lastSpawn + 1500) { spawnEnemy(this); lastSpawn = time; } // ---- ENEMIGOS SE MUEVEN ---- enemies.children.iterate(e => { if (e) e.setVelocityY(100); }); // ---- LIMPIEZA (fuera de pantalla) ---- bullets.children.iterate(b => { if (b && b.y < -50) b.setActive(false).setVisible(false); }); enemies.children.iterate(e => { if (e && e.y > config.height + 50) { e.setActive(false).setVisible(false); } }); } // ------------------------------------------------- function spawnEnemy(scene) { const x = Phaser.Math.Between(30, config.width - 30); enemy.disableBody(true, true); score += 10; scoreText.setText('Puntos: ' + score); } // ------------------------------------------------- function onPlayerHit(player, enemy) { this.physics.pause(); player.setTint(0xff0000); const txt = this.add.text(config.width/2, config.height/2, 'Game Over\nPuntuación: ' + score, { font: '32px monospace', fill: '#fff', align: 'center' }) .setOrigin(0.5); // Reinicio rápido con clic this.input.once('pointerdown', () => { this.scene.restart(); }); }
